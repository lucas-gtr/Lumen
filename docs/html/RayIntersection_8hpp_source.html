67inline bool getTriangleIntersection(const Ray& ray, const linalg::Vec3d& p0, const linalg::Vec3d& p1,
68 const linalg::Vec3d& p2, double& hit_distance, linalg::Vec3d& bary_coords) {
69 const linalg::Vec3d edge1 = p1 - p0;
70 const linalg::Vec3d edge2 = p2 - p0;
71 const linalg::Vec3d h = ray.direction.cross(edge2);
72 const double a = linalg::dot(edge1, h);
73
74 if(a > -INTERSECTION_TOLERANCE && a < INTERSECTION_TOLERANCE) {
75 return false;
76 }
77
78 const double f = 1.0 / a;
79 const linalg::Vec3d s = ray.origin - p0;
80 const double u = f * linalg::dot(s, h);
81
82 if(u < -INTERSECTION_TOLERANCE || u > 1.0 + INTERSECTION_TOLERANCE) {
83 return false;
84 }
85
86 const linalg::Vec3d q = s.cross(edge1);
87 const double v = f * linalg::dot(ray.direction, q);
88
89 if(v < -INTERSECTION_TOLERANCE || u + v > 1.0 + INTERSECTION_TOLERANCE) {
90 return false;
91 }
92
93 hit_distance = f * linalg::dot(edge2, q);
94
95 if(hit_distance > INTERSECTION_TOLERANCE) {
96 bary_coords = {1.0 - u - v, u, v};
97 return true;
98 }
99
100 return false;
101}

168inline bool getAABBIntersection(const linalg::Vec3d& origin, const linalg::Vec3d& inv_dir,
169 const linalg::Vec3d& min_bound, const linalg::Vec3d& max_bound, double& hit_distance) {
170 const linalg::Vec3d t0 = linalg::cwiseProduct((min_bound - origin), inv_dir);
171 const linalg::Vec3d t1 = linalg::cwiseProduct((max_bound - origin), inv_dir);
172
173 const linalg::Vec3d t_min = linalg::cwiseMin(t0, t1);
174 const linalg::Vec3d t_max = linalg::cwiseMax(t0, t1);
175
176 const double t_entry = t_min.maxValue();
177 const double t_exit = t_max.minValue();
178
179 if(t_entry > t_exit || t_exit < 0) {
180 return false;
181 }
182 hit_distance = t_entry;
183 return true;
184}

220inline void updateHitInfoFromBarycentric(RayHitInfo& hit_info, double distance, const linalg::Vec3d& bary,
221 const Vertex& v0, const Vertex& v1, const Vertex& v2) {
222 hit_info.distance = distance;
223 hit_info.bary_coords.u = bary.x * v0.uv_coord.u + bary.y * v1.uv_coord.u + bary.z * v2.uv_coord.u;
224 hit_info.bary_coords.v = bary.x * v0.uv_coord.v + bary.y * v1.uv_coord.v + bary.z * v2.uv_coord.v;
225
226 hit_info.normal = (bary.x * v0.normal + bary.y * v1.normal + bary.z * v2.normal).normalized();
227 hit_info.tangent = (bary.x * v0.tangent + bary.y * v1.tangent + bary.z * v2.tangent).normalized();
228 hit_info.bitangent = (bary.x * v0.bitangent + bary.y * v1.bitangent + bary.z * v2.bitangent).normalized();
229 hit_info.area = (v1.position - v0.position).cross(v2.position - v0.position).length() * HALF;
230}
237inline Ray transformRayToObjectSpace(const Ray& ray, const Object3D* object) {

239
240 const linalg::Vec3d origin = linalg::toVec3((inv_matrix * linalg::toVec4(ray.origin)));
241 const linalg::Vec3d direction = (inv_matrix.topLeft3x3() * ray.direction).normalized();

253inline void transformHitInfoToWorldSpace(RayHitInfo& hit_info, const Ray& local_ray, const Ray& original_ray,
254 const Object3D* object) {
255 const linalg::Vec3d local_position = local_ray.origin + hit_info.distance * local_ray.direction;
256 const linalg::Vec3d world_position =
257 linalg::toVec3((object->getTransformationMatrix() * linalg::toVec4(local_position)));
258
259 hit_info.distance = (world_position - original_ray.origin).length();
260 hit_info.position = world_position;
261 hit_info.material = object->getMaterial();
262
263 hit_info.emitted_light =
264 hit_info.material->getEmissive(hit_info.bary_coords) * hit_info.material->getEmissiveIntensity();
265
266 const linalg::Mat3d rotation_scale_matrix = object->getTransformationMatrix().topLeft3x3();
267 const linalg::Vec3d tangent_world = rotation_scale_matrix * hit_info.tangent;
268 const linalg::Vec3d bitangent_world = rotation_scale_matrix * hit_info.bitangent;
269 const double area_scale = tangent_world.cross(bitangent_world).length();
270 hit_info.area *= area_scale;
271
272 const linalg::Mat3d normal_matrix = object->getNormalMatrix();
273 hit_info.normal = (normal_matrix * hit_info.normal).normalized();
274 hit_info.tangent = (normal_matrix * hit_info.tangent).normalized();
275 hit_info.bitangent = (normal_matrix * hit_info.bitangent).normalized();
276}
